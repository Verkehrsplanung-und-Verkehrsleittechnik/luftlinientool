21def take_screenshot(name_screenshot, gpa_path: str = ""):
22 """
23 This function creates a screenshot of the current Visum network.
24 If a path to saved graphic parameters is provided, they will be loaded.
25 The file is saved under the name `name_screenshot` as a .jpg file.
26 """
27 if gpa_path != "":
28 # Load the saved graphic parameters if a path is specified
29 Visum.Net.GraphicParameters.Open(Path.cwd() / gpa_path)
30
31 # The entire network is loaded into the viewport and displayed
32 Visum.Graphic.DisplayEntireNetwork()
33
34 # The screenshot is created and saved as a .jpg file
35 jpg_path = Path(Visum.GetPath(41)) / f"{name_screenshot}.jpg"
36 Visum.Graphic.Screenshot(jpg_path, 3) # Parameter 3 specifies the quality level
37 return jpg_path

62def filter_zones_source(attr_source):
63 """
64 This function filters zones based on a given attribute.
65 Only zones where the attribute is greater than 0 are considered.
66 """
67 filter = Visum.Filters.ZoneFilter()
68 filter.Init()
69 if attr_source is not None:
70 # Set the condition for the filter: The attribute must be greater than 0
71 filter.AddCondition("OP_NONE", False, attr_source, "GreaterVal", 0)
72 filter.UseFilter = True # Activate the filter

77def filter_links_vfs(set_linktypes):
78 """
79 This function filters the links based on the given link types.
80 Only links whose type is contained in `set_linktypes` are considered.
81 """
82 filter = Visum.Filters.LinkFilter()
83 filter.Init()
84 # Filter links by their type (TypeNo), based on the given set of link types
85 filter.AddCondition("OP_NONE", False, "TypeNo", "ContainedIn", ",".join(str(x) for x in set_linktypes))
86 filter.UseFilter = True # Activate the filter
